```glsl
in  vec2 v_UV;
out vec4 FragColor;

uniform sampler2D u_Source;
uniform vec2  u_TexelSize;    
uniform int   u_FirstPass;
uniform float u_Threshold;

vec3 Downsample13(vec2 uv, vec2 texel)
{
    const vec3 a = texture(u_Source, uv + texel * vec2(-2.0, -2.0)).rgb;
    const vec3 b = texture(u_Source, uv + texel * vec2( 0.0, -2.0)).rgb;
    const vec3 c = texture(u_Source, uv + texel * vec2( 2.0, -2.0)).rgb;
    const vec3 d = texture(u_Source, uv + texel * vec2(-2.0,  0.0)).rgb;
    const vec3 e = texture(u_Source, uv).rgb;
    const vec3 f = texture(u_Source, uv + texel * vec2( 2.0,  0.0)).rgb;
    const vec3 g = texture(u_Source, uv + texel * vec2(-2.0,  2.0)).rgb;
    const vec3 h = texture(u_Source, uv + texel * vec2( 0.0,  2.0)).rgb;
    const vec3 i = texture(u_Source, uv + texel * vec2( 2.0,  2.0)).rgb;

    const vec3 j = texture(u_Source, uv + texel * vec2(-1.0, -1.0)).rgb;
    const vec3 k = texture(u_Source, uv + texel * vec2( 1.0, -1.0)).rgb;
    const vec3 l = texture(u_Source, uv + texel * vec2(-1.0,  1.0)).rgb;
    const vec3 m = texture(u_Source, uv + texel * vec2( 1.0,  1.0)).rgb;

    vec3 result  = e * 0.125;
    result += (a + c + g + i) * 0.03125;
    result += (b + d + f + h) * 0.0625;
    result += (j + k + l + m) * 0.125;
    return result;
}

vec3 BrightPass(vec3 c, float threshold)
{
    const float br = max(c.r, max(c.g, c.b));
    const float knee = max(threshold * 0.5, 1e-4);

    float soft = clamp(br - threshold + knee, 0.0, 2.0 * knee);
    soft = soft * soft / (4.0 * knee);

    const float contrib = max(soft, br - threshold) / max(br, 1e-4);
    return c * contrib;
}

void main()
{
    vec3 c = Downsample13(v_UV, u_TexelSize);

    if (u_FirstPass != 0)
        c = BrightPass(c, u_Threshold);

    FragColor = vec4(c, 1.0);
}
```

```glsl
#version 460 core

layout(local_size_x = 64, local_size_y = 1, local_size_z = 1) in;

struct InstanceGPU {
    mat4 Model;
    mat4 PrevModel;
    mat4 NormalMatrix;   
    uint MaterialId;
    uint MeshId;
    uint BoneOffset;
    uint Flags;
};

struct MeshGPU {
    uint  BaseVertex;
    uint  VertexCount;
    uint  BaseIndex;
    uint  IndexCount;
    vec3  AABBMin;
    float _pad0;
    vec3  AABBMax;
    float _pad1;
};

struct DrawElementsIndirect {
    uint Count;
    uint InstanceCount;
    uint FirstIndex;
    uint BaseVertex;
    uint BaseInstance;
};

layout(std430, binding = 1) readonly buffer Instances {
    InstanceGPU instances[];
};

layout(std430, binding = 5) readonly buffer Meshes {
    MeshGPU meshes[];
};

layout(std430, binding = 6) writeonly buffer DrawCommands {
    DrawElementsIndirect draws[];
};

layout(std430, binding = 7) buffer DrawState {
    uint drawCount;
    uint vertexCount;
    uint indexCount;
    uint visibleInstanceCount;
};

layout(std430, binding = 8) readonly buffer GroupTemplates {
    DrawElementsIndirect groupTemplates[];
};

uniform vec4 u_Frustum[6];
uniform uint u_GroupCount;
uniform uint u_MaxDraws;

const uint kInstanceFlagSkinned = 1u;

shared uint s_anyVisible;
shared uint s_visibleCount;   

void main()
{
    const uint groupIdx = gl_WorkGroupID.x;
    if (groupIdx >= u_GroupCount)
        return;

    const DrawElementsIndirect tpl = groupTemplates[groupIdx];
    const uint base  = tpl.BaseInstance;
    const uint count = tpl.InstanceCount;
    if (count == 0u)
        return;

    const uint lane   = gl_LocalInvocationID.x;
    const uint stride = gl_WorkGroupSize.x;

    if (lane == 0u) {
        s_anyVisible   = 0u;
        s_visibleCount = 0u;
    }
    barrier();

    for (uint i = lane; i < count; i += stride)
    {
        const uint instanceIndex = base + i;
        const InstanceGPU inst = instances[instanceIndex];
        const MeshGPU    mesh = meshes[inst.MeshId];

        bool visible = ((inst.Flags & kInstanceFlagSkinned) != 0u);

        if (!visible)
        {
            const vec3 localCenter = (mesh.AABBMin + mesh.AABBMax) * 0.5;
            const vec3 localExtent = (mesh.AABBMax - mesh.AABBMin) * 0.5;
            const vec3 worldCenter = (inst.Model * vec4(localCenter, 1.0)).xyz;
            const mat3 absBasis = mat3(abs(inst.Model[0].xyz),
                                       abs(inst.Model[1].xyz),
                                       abs(inst.Model[2].xyz));
            const vec3 worldExtent = absBasis * localExtent;

            visible = true;
            for (int p = 0; p < 6; ++p)
            {
                const vec3  n = u_Frustum[p].xyz;
                const float d = u_Frustum[p].w;
                if (dot(n, worldCenter) + d < -dot(abs(n), worldExtent))
                {
                    visible = false;
                    break;
                }
            }
        }

        if (visible) {
            atomicOr(s_anyVisible, 1u);
            atomicAdd(s_visibleCount, 1u);   
        }
    }
    barrier();

    if (lane == 0u && s_anyVisible != 0u)
    {
        const uint slot = atomicAdd(drawCount, 1u);
        if (slot >= u_MaxDraws)
            return;

        draws[slot] = tpl;

        const MeshGPU m = meshes[instances[base].MeshId];

        
        atomicAdd(vertexCount,          m.VertexCount * s_visibleCount);
        atomicAdd(indexCount,           m.IndexCount  * s_visibleCount);
        atomicAdd(visibleInstanceCount, s_visibleCount);
    }
}
```

```glsl
#version 460 core




in vec2 v_UV;
out vec4 FragColor;

uniform sampler2D u_ColorTexture;

uniform vec2  u_TexelSize;          
uniform float u_EdgeThreshold;      
uniform float u_EdgeThresholdMin;   
uniform float u_SubpixelBlending;   

float Luma(vec3 rgb)
{
    return dot(rgb, vec3(0.299, 0.587, 0.114));
}

void main()
{
    vec3 rgbM  = texture(u_ColorTexture, v_UV).rgb;
    vec3 rgbNW = textureOffset(u_ColorTexture, v_UV, ivec2(-1, -1)).rgb;
    vec3 rgbNE = textureOffset(u_ColorTexture, v_UV, ivec2( 1, -1)).rgb;
    vec3 rgbSW = textureOffset(u_ColorTexture, v_UV, ivec2(-1,  1)).rgb;
    vec3 rgbSE = textureOffset(u_ColorTexture, v_UV, ivec2( 1,  1)).rgb;

    float lumaM  = Luma(rgbM);
    float lumaNW = Luma(rgbNW);
    float lumaNE = Luma(rgbNE);
    float lumaSW = Luma(rgbSW);
    float lumaSE = Luma(rgbSE);

    float lumaMin = min(lumaM, min(min(lumaNW, lumaNE), min(lumaSW, lumaSE)));
    float lumaMax = max(lumaM, max(max(lumaNW, lumaNE), max(lumaSW, lumaSE)));
    float lumaRange = lumaMax - lumaMin;

    
    if (lumaRange < max(u_EdgeThresholdMin, lumaMax * u_EdgeThreshold))
    {
        FragColor = vec4(rgbM, 1.0);
        return;
    }

    
    float lumaAvg = (lumaNW + lumaNE + lumaSW + lumaSE + lumaM * 4.0) * 0.125;
    float subpixel = clamp(abs(lumaAvg - lumaM) / max(lumaRange, 1e-4), 0.0, 1.0);
    subpixel = subpixel * subpixel * u_SubpixelBlending;

    
    vec2 dir;
    dir.x = -((lumaNW + lumaNE) - (lumaSW + lumaSE));
    dir.y =  ((lumaNW + lumaSW) - (lumaNE + lumaSE));

    float dirReduce = max((lumaNW + lumaNE + lumaSW + lumaSE) * 0.25 * (1.0 / 8.0),
                          1.0 / 128.0);
    float rcpDirMin = 1.0 / (min(abs(dir.x), abs(dir.y)) + dirReduce);
    dir = clamp(dir * rcpDirMin, vec2(-8.0), vec2(8.0)) * u_TexelSize;

    
    vec3 rgbA = 0.5 * (
        texture(u_ColorTexture, v_UV + dir * (1.0 / 3.0 - 0.5)).rgb +
        texture(u_ColorTexture, v_UV + dir * (2.0 / 3.0 - 0.5)).rgb);

    vec3 rgbB = rgbA * 0.5 + 0.25 * (
        texture(u_ColorTexture, v_UV + dir * -0.5).rgb +
        texture(u_ColorTexture, v_UV + dir *  0.5).rgb);

    float lumaB = Luma(rgbB);
    vec3 result = ((lumaB < lumaMin) || (lumaB > lumaMax)) ? rgbA : rgbB;

    result = mix(result, rgbM, subpixel);

    FragColor = vec4(result, 1.0);
}
```

```glsl
layout(location = 1) out vec3  g_Normal;
layout(location = 2) out vec4  g_Albedo_AO;
layout(location = 3) out vec4  g_Material;
layout(location = 4) out vec2  g_Velocity;

in vec3 v_Normal;

in vec2 v_Velocity;
flat in uint v_InstanceIndex;

struct InstanceGPU {
    mat4 Model;
    mat4 PrevModel;
    mat4 NormalMatrix;   
    uint MaterialId;
    uint MeshId;
    uint BoneOffset;
    uint Flags;
};

layout(std430, binding = 1) readonly buffer Instances {
    InstanceGPU instances[];
};

struct MaterialGPU {
    vec4  AlbedoRoughness;
    vec4  MetallicAOEms;
    vec4  SkyReflParams;
    vec4  StrengthA;
    uvec2 AlbedoHandle;
    uvec2 NormalHandle;
    uvec2 MetallicHandle;
    uvec2 RoughnessHandle;
    uvec2 AOHandle;
    uvec2 EmissiveHandle;
    vec4  StrengthB;
};

layout(std430, binding = 2) readonly buffer Materials {
    MaterialGPU materials[];
};

vec3 sRGBToLinear(vec3 c) { return pow(c, vec3(2.2)); }

mat3 ComputeTBN(vec3 N, vec3 worldPos, vec2 uv)
{
    vec3 dp1  = dFdx(worldPos);
    vec3 dp2  = dFdy(worldPos);
    vec2 duv1 = dFdx(uv);
    vec2 duv2 = dFdy(uv);

    vec3 dp2perp = cross(dp2, N);
    vec3 dp1perp = cross(N, dp1);

    vec3 T = dp2perp * duv1.x + dp1perp * duv2.x;
    vec3 B = dp2perp * duv1.y + dp1perp * duv2.y;

    float len2 = max(dot(T, T), dot(B, B));
    if (len2 < 1e-12)
        return mat3(1.0);

    float invmax = inversesqrt(len2);
    return mat3(T * invmax, B * invmax, N);
}

const uvec2 kNoHandle = uvec2(0u, 0u);

void main()
{
    InstanceGPU inst = instances[v_InstanceIndex];
    MaterialGPU m    = materials[inst.MaterialId];

    vec2 uv = v_UV * m.MetallicAOEms.w;

    vec3 albedoTint = sRGBToLinear(m.AlbedoRoughness.rgb);
    vec3 baseColor  = albedoTint;

    if (m.AlbedoHandle != kNoHandle)
        baseColor *= texture(sampler2D(m.AlbedoHandle), uv).rgb * m.StrengthB.x;

    vec3 N = normalize(v_Normal);

    if (m.NormalHandle != kNoHandle)
    {
        mat3 TBN = ComputeTBN(N, v_WorldPos, uv);
        vec3 s   = texture(sampler2D(m.NormalHandle), uv).rgb * 2.0 - 1.0;
        s.xy    *= m.StrengthB.y;
        N        = normalize(TBN * normalize(s));
    }

    float metallic  = m.MetallicAOEms.x;
    float roughness = m.AlbedoRoughness.w;
    float ao        = m.MetallicAOEms.y;
    float emissive  = m.MetallicAOEms.z;

    if (m.MetallicHandle != kNoHandle)
        metallic = clamp(texture(sampler2D(m.MetallicHandle), uv).r * m.StrengthA.x, 0.0, 1.0);

    if (m.RoughnessHandle != kNoHandle)
        roughness = clamp(texture(sampler2D(m.RoughnessHandle), uv).r * m.StrengthA.y, 0.04, 1.0);

    if (m.AOHandle != kNoHandle)
        ao = clamp(texture(sampler2D(m.AOHandle), uv).r * m.StrengthA.z, 0.0, 1.0);

    if (m.EmissiveHandle != kNoHandle)
        emissive = texture(sampler2D(m.EmissiveHandle), uv).r * m.StrengthA.w;

    float skyReflPacked = (m.SkyReflParams.x > 0.5)
        ? max(m.SkyReflParams.y, 0.0)
        : -1.0;

    g_Position  = v_WorldPos;
    g_Normal    = N;
    g_Albedo_AO = vec4(baseColor, ao);
    g_Material  = vec4(metallic, roughness, emissive, skyReflPacked);
    g_Velocity  = v_Velocity;
}
```

```glsl
layout(location = 1) in vec3  a_Normal;
layout(location = 2) in vec2  a_UV;
layout(location = 3) in uvec4 a_BoneIndices;

    mat4 PrevModel;
    mat4 NormalMatrix;

    uint MeshId;
    uint BoneOffset;
    uint Flags;
};

    InstanceGPU instances[];
};

    mat4 bones[];
};

uniform mat4 u_View;
uniform mat4 u_Projection;
uniform mat4 u_PrevViewProj;

out vec3 v_WorldPos;
out vec3 v_Normal;
out vec2 v_UV;
out vec2 v_Velocity;
flat out uint v_InstanceIndex;

void main()
{
    uint instanceId = uint(gl_BaseInstance) + uint(gl_InstanceID);
    InstanceGPU inst = instances[instanceId];

    vec4 localPos    = vec4(a_Position, 1.0);
    vec3 localNormal = a_Normal;

    if ((inst.Flags & 1u) != 0u &&
        (a_BoneWeights.x + a_BoneWeights.y + a_BoneWeights.z + a_BoneWeights.w) > 0.0)
    {
        mat4 skin =
            bones[inst.BoneOffset + a_BoneIndices.x] * a_BoneWeights.x +
            bones[inst.BoneOffset + a_BoneIndices.y] * a_BoneWeights.y +
            bones[inst.BoneOffset + a_BoneIndices.z] * a_BoneWeights.z +
            bones[inst.BoneOffset + a_BoneIndices.w] * a_BoneWeights.w;
        localPos    = skin * localPos;
        localNormal = mat3(skin) * localNormal;
    }

    vec4 worldPos = inst.Model * localPos;

    vec4 currClip = u_Projection * u_View * worldPos;
    vec4 prevClip = u_PrevViewProj * (inst.PrevModel * localPos);

    vec2 velocity = vec2(0.0);
    if (prevClip.w > 1e-5 && currClip.w > 1e-5)
        velocity = (prevClip.xy / prevClip.w - currClip.xy / currClip.w) * 0.5;

    v_Velocity = velocity;
    v_WorldPos = worldPos.xyz;

    
    
    v_Normal = normalize(mat3(inst.NormalMatrix) * localNormal);

    v_UV            = a_UV;
    v_InstanceIndex = instanceId;
    gl_Position     = currClip;
}
```

```glsl
#version 460 core

in vec2 v_UV;
out vec4 FragColor;

uniform sampler2D u_PositionTexture;
uniform sampler2D u_NormalTexture;
uniform sampler2D u_AlbedoTexture;
uniform sampler2D u_MaterialTexture;

uniform vec3  u_CameraPosition;
uniform mat4  u_InvViewProj;


uniform samplerCube u_EnvironmentMap;
uniform int   u_HasEnvironment;
uniform float u_SkyboxRotation;    
uniform float u_SkyboxBrightness;


uniform int   u_FogEnabled;
uniform int   u_FogMode;        
uniform vec3  u_FogColor;       
uniform float u_FogDensity;
uniform float u_FogStart;
uniform float u_FogEnd;

const int MAX_LIGHTS = 32;
struct Light {
    vec3  Position;
    float Range;
    vec3  Direction;
    float Intensity;
    vec3  Color;
    int   LightType;
    float InnerCutoff;
    float OuterCutoff;
};
uniform Light u_Lights[MAX_LIGHTS];
uniform int   u_LightCount;

const float PI = 3.14159265359;



float D_GGX(float NdotH, float roughness)
{
    float a  = roughness * roughness;
    float a2 = a * a;
    float d  = NdotH * NdotH * (a2 - 1.0) + 1.0;
    return a2 / max(PI * d * d, 1e-6);
}

float G_SchlickGGX(float NdotX, float k)
{
    return NdotX / (NdotX * (1.0 - k) + k);
}

float G_Smith(float NdotV, float NdotL, float roughness)
{
    float r = roughness + 1.0;
    float k = (r * r) / 8.0;
    return G_SchlickGGX(NdotV, k) * G_SchlickGGX(NdotL, k);
}

vec3 F_Schlick(float cosTheta, vec3 F0)
{
    return F0 + (1.0 - F0) * pow(1.0 - cosTheta, 5.0);
}

vec3 EvaluateBRDF(vec3 N, vec3 V, vec3 L, vec3 radiance,
                  vec3 albedo, float metallic, float roughness)
{
    
    vec3 Hv = V + L;
    float hLen = length(Hv);
    if (hLen < 1e-5) return vec3(0.0);
    vec3 H = Hv / hLen;

    float NdotV = max(dot(N, V), 1e-4);
    float NdotL = max(dot(N, L), 0.0);
    float NdotH = max(dot(N, H), 0.0);
    float HdotV = max(dot(H, V), 0.0);

    if (NdotL <= 0.0) return vec3(0.0);

    vec3 F0 = mix(vec3(0.04), albedo, metallic);

    float D = D_GGX(NdotH, roughness);
    float G = G_Smith(NdotV, NdotL, roughness);
    vec3  F = F_Schlick(HdotV, F0);

    vec3 specular = (D * G * F) / max(4.0 * NdotV * NdotL, 1e-4);

    vec3 kD = (1.0 - F) * (1.0 - metallic);
    vec3 diffuse = kD * albedo / PI;

    return (diffuse + specular) * radiance * NdotL;
}

float PointAttenuation(float dist, float range)
{
    float d2 = dist * dist;
    float t  = dist / max(range, 1e-3);
    float cutoff = pow(clamp(1.0 - pow(t, 4.0), 0.0, 1.0), 2.0);
    return cutoff / max(d2, 1e-3);
}



vec3 ComputeRayDirection()
{
    
    vec4 clip  = vec4(v_UV * 2.0 - 1.0, 1.0, 1.0);
    vec4 world = u_InvViewProj * clip;
    return normalize(world.xyz / world.w - u_CameraPosition);
}

vec3 SampleSkybox(vec3 dir)
{
    
    float c = cos(u_SkyboxRotation);
    float s = sin(u_SkyboxRotation);
    vec3 rotated = vec3(
        dir.x * c - dir.z * s,
        dir.y,
        dir.x * s + dir.z * c);

    return texture(u_EnvironmentMap, rotated).rgb * u_SkyboxBrightness;
}



void main()
{
    vec3 worldPos = texture(u_PositionTexture, v_UV).rgb;
    vec3 Nraw     = texture(u_NormalTexture,   v_UV).rgb;

    vec4 albedoAO = texture(u_AlbedoTexture, v_UV);
    vec3 albedo   = albedoAO.rgb;
    float ao      = albedoAO.a;

    vec4 mat        = texture(u_MaterialTexture, v_UV);
    float metallic  = mat.r;
    float roughness = clamp(mat.g, 0.04, 1.0);
    float emissive  = mat.b;

    
    bool  skyReflEnabled  = mat.a >= 0.0;
    float skyReflStrength = max(mat.a, 0.0);

    
    
    
    if (dot(Nraw, Nraw) < 1e-4)
    {
        vec3 color = vec3(0.0);

        if (u_HasEnvironment == 1)
        {
            vec3 dir = ComputeRayDirection();
            color = SampleSkybox(dir);
        }

        
        FragColor = vec4(color, 1.0);
        return;
    }

    
    vec3 N = normalize(Nraw);
    vec3 V = normalize(u_CameraPosition - worldPos);
    vec3 Lo = vec3(0.0);

    for (int i = 0; i < u_LightCount; ++i)
    {
        vec3 L;
        vec3 radiance;

        if (u_Lights[i].LightType == 0)
        {
            L = normalize(-u_Lights[i].Direction);
            radiance = u_Lights[i].Color * u_Lights[i].Intensity;
        }
        else if (u_Lights[i].LightType == 1)
        {
            vec3  toLight = u_Lights[i].Position - worldPos;
            float dist    = length(toLight);
            L = toLight / max(dist, 1e-4);
            radiance = u_Lights[i].Color * u_Lights[i].Intensity
                     * PointAttenuation(dist, u_Lights[i].Range);
        }
        else
        {
            vec3  toLight = u_Lights[i].Position - worldPos;
            float dist    = length(toLight);
            L = toLight / max(dist, 1e-4);

            float theta   = dot(L, normalize(-u_Lights[i].Direction));
            float epsilon = u_Lights[i].InnerCutoff - u_Lights[i].OuterCutoff;
            float spot    = clamp((theta - u_Lights[i].OuterCutoff)
                                  / max(epsilon, 1e-4), 0.0, 1.0);
            spot *= spot;

            radiance = u_Lights[i].Color * u_Lights[i].Intensity
                     * PointAttenuation(dist, u_Lights[i].Range)
                     * spot;
        }

        Lo += EvaluateBRDF(N, V, L, radiance, albedo, metallic, roughness);
    }

    
    vec3 skyColor    = vec3(0.35, 0.42, 0.55);
    vec3 groundColor = vec3(0.12, 0.10, 0.08);
    float hemi       = N.y * 0.5 + 0.5;
    vec3 ambient     = mix(groundColor, skyColor, hemi) * albedo * ao;

    if (u_HasEnvironment == 1)
    {
        
        vec3 skyN = SampleSkybox(N);
        ambient += skyN * albedo * ao * 0.15;

        
        if (skyReflEnabled)
        {
            vec3 R    = reflect(-V, N);
            vec3 skyR = SampleSkybox(R);

            vec3 reflTint = mix(vec3(1.0), albedo, metallic);

            float NdotV   = max(dot(N, V), 1e-4);
            float graze   = pow(1.0 - NdotV, 5.0);
            float fresnel = mix(1.0, 1.6, graze);

            ambient += skyR * reflTint * fresnel * skyReflStrength * ao;
        }
    }

    vec3 color = ambient + Lo + albedo * emissive;

    
    
    
    if (u_FogEnabled == 1)
    {
        const float dist = length(u_CameraPosition - worldPos);

        float fog = 0.0;
        if (u_FogMode == 0)
        {
            fog = clamp((dist - u_FogStart) / max(u_FogEnd - u_FogStart, 1e-4),
                        0.0, 1.0);
        }
        else if (u_FogMode == 1)
        {
            fog = 1.0 - exp(-max(dist, 0.0) * max(u_FogDensity, 0.0));
        }
        else
        {
            const float d = max(dist, 0.0) * max(u_FogDensity, 0.0);
            fog = 1.0 - exp(-d * d);
        }

        
        const vec3 fogLinear = pow(max(u_FogColor, vec3(0.0)), vec3(2.2));
        color = mix(color, fogLinear, clamp(fog, 0.0, 1.0));
    }

    
    
    FragColor = vec4(color, 1.0);
}
```

```glsl
in  vec2 v_UV;
out vec4 FragColor;

uniform sampler2D u_HDRTexture;
uniform sampler2D u_VelocityTexture;
uniform sampler2D u_DepthTexture;

uniform vec2  u_JitterDelta;   
uniform float u_Strength;
uniform int   u_Samples;
uniform float u_NearPlane;
uniform float u_FarPlane;

float LinearDepth(float d)
{
    return u_NearPlane * u_FarPlane /
           max(u_FarPlane - d * (u_FarPlane - u_NearPlane), 1e-6);
}

void main()
{
    const vec3 center = texture(u_HDRTexture, v_UV).rgb;

    vec2 vel = texture(u_VelocityTexture, v_UV).rg - u_JitterDelta;
    vel *= u_Strength;

    const vec2 res = vec2(textureSize(u_HDRTexture, 0));
    const float lenPx = length(vel * res);

    
    if (lenPx < 0.5)
    {
        FragColor = vec4(center, 1.0);
        return;
    }

    
    const float kMaxPixels = 32.0;
    if (lenPx > kMaxPixels)
        vel *= kMaxPixels / lenPx;

    const float zc = LinearDepth(texture(u_DepthTexture, v_UV).r);

    vec3 sum  = center;
    float wsum = 1.0;

    const int samples = clamp(u_Samples, 2, 16);
    for (int i = 1; i < samples; ++i)
    {
        const float t = float(i) / float(samples - 1) - 0.5;   
        const vec2 suv = v_UV + vel * t;

        if (any(lessThan(suv, vec2(0.0))) || any(greaterThan(suv, vec2(1.0))))
            continue;

        const float zs = LinearDepth(texture(u_DepthTexture, suv).r);
        if (abs(zs - zc) > 0.1 * zc)
            continue;   

        sum  += texture(u_HDRTexture, suv).rgb;
        wsum += 1.0;
    }

    FragColor = vec4(sum / wsum, 1.0);
}
```

```glsl
#version 460 core

in  vec2 v_UV;
out vec4 FragColor;

uniform sampler2D u_CurrentTexture;
uniform sampler2D u_VelocityTexture;
uniform sampler2D u_HistoryTexture;
uniform sampler2D u_DepthTexture;

uniform vec2  u_TexelSize;
uniform vec2  u_Resolution;
uniform float u_HistoryWeight;
uniform int   u_HistoryValid;
uniform float u_ClampGamma;

uniform mat4  u_ViewProj;
uniform mat4  u_PrevViewProj;
uniform mat4  u_InvViewProj;
uniform vec3  u_CameraPosition;

vec3 RGBToYCoCg(vec3 c)
{
    return vec3(dot(c, vec3(0.25,  0.5,  0.25)),
                dot(c, vec3(0.5,   0.0, -0.5)),
                dot(c, vec3(-0.25, 0.5, -0.25)));
}

vec3 YCoCgToRGB(vec3 c)
{
    return vec3(c.x + c.y - c.z,
                c.x + c.z,
                c.x - c.y - c.z);
}

bool IsFinite(vec3 v)
{
    const bvec3 isNan = notEqual(v, v);
    const bvec3 isInf = greaterThanEqual(abs(v), vec3(1e30));
    return !(any(isNan) || any(isInf));
}

vec2 SkyVelocity(vec2 uv)
{
    const vec4 far = u_InvViewProj * vec4(uv * 2.0 - 1.0, 1.0, 1.0);
    const vec3 dir = normalize(far.xyz / far.w - u_CameraPosition);

    const vec4 prevClip = u_PrevViewProj * vec4(dir, 0.0);
    const vec4 currClip = u_ViewProj     * vec4(dir, 0.0);

    if (prevClip.w <= 1e-5 || currClip.w <= 1e-5)
        return vec2(0.0);

    return (prevClip.xy / prevClip.w - currClip.xy / currClip.w) * 0.5;
}

void main()
{
    const vec3  current = texture(u_CurrentTexture, v_UV).rgb;
    const float depth   = texture(u_DepthTexture, v_UV).r;

    vec2 vel = texture(u_VelocityTexture, v_UV).rg;
    if (depth >= 0.999999)
        vel = SkyVelocity(v_UV);

    if (any(isnan(vel)) || any(isinf(vel)))
        vel = vec2(0.0);

    
    
    
    
    const vec2 prevUV = v_UV + vel;

    if (u_HistoryValid == 0 ||
        !IsFinite(current) ||
        any(lessThan(prevUV, vec2(0.0))) ||
        any(greaterThan(prevUV, vec2(1.0))))
    {
        FragColor = vec4(current, 1.0);
        return;
    }

    const vec3 curY = RGBToYCoCg(max(current, vec3(0.0)));

    vec3 m1 = vec3(0.0);
    vec3 m2 = vec3(0.0);
    for (int y = -1; y <= 1; ++y)
    {
        for (int x = -1; x <= 1; ++x)
        {
            const vec3 s = RGBToYCoCg(max(texture(u_CurrentTexture,
                                                  v_UV + vec2(x, y) * u_TexelSize).rgb,
                                           vec3(0.0)));
            m1 += s;
            m2 += s * s;
        }
    }
    m1 /= 9.0;
    m2 /= 9.0;

    const vec3 sigma = sqrt(max(m2 - m1 * m1, vec3(0.0)));
    vec3 lo = m1 - u_ClampGamma * sigma;
    vec3 hi = m1 + u_ClampGamma * sigma;
    lo = min(lo, curY);
    hi = max(hi, curY);

    const vec3 historyY = RGBToYCoCg(max(texture(u_HistoryTexture, prevUV).rgb,
                                         vec3(0.0)));
    const vec3 clampedHistory = clamp(historyY, lo, hi);

    const float speedPx = length(vel * u_Resolution);
    float weight = u_HistoryWeight;
    weight *= 1.0 - smoothstep(24.0, 96.0, speedPx);
    weight = clamp(weight, 0.0, 0.98);

    FragColor = vec4(YCoCgToRGB(mix(curY, clampedHistory, weight)), 1.0);
}
```

```glsl
uniform float u_VignetteIntensity;

vec3 TonemapACES(vec3 x)
{
    const float a = 2.51;
    const float b = 0.03;
    const float c = 2.43;
    const float d = 0.59;
    const float e = 0.14;
    return clamp((x * (a * x + b)) / (x * (c * x + d) + e), 0.0, 1.0);
}

vec3 TonemapReinhard(vec3 x)
{
    return x / (1.0 + x);
}


vec3 TonemapFilmic(vec3 x)
{
    const float A = 0.15;
    const float B = 0.50;
    const float C = 0.10;
    const float D = 0.20;
    const float E = 0.02;
    const float F = 0.30;

    const vec3 curve = ((x * (A * x + C * B) + D * E) /
                        (x * (A * x + B) + D * F)) - E / F;

    const float W = 11.2;
    const float white = ((W * (A * W + C * B) + D * E) /
                         (W * (A * W + B) + D * F)) - E / F;
    return curve / white;
}



vec3 TonemapAgX(vec3 x)
{
    const mat3 agxIn = mat3(
        0.842479062253094, 0.0784335999999992, 0.0792237451477643,
        0.0423282422610123, 0.878468636469772, 0.0791661274605434,
        0.0423756549057051, 0.0784336, 0.879142973793104);

    const mat3 agxOut = mat3(
         1.19687900512017, -0.0980208811401368, -0.0990297440797205,
        -0.0528968517574562, 1.15190312990417, -0.0989611768448433,
        -0.0529716355144438, -0.0980434501171241, 1.15107367264116);

    const float minEv = -12.47393;
    const float maxEv = 4.026069;

    x = agxIn * max(x, vec3(0.0));
    x = clamp(log2(max(x, vec3(1e-10))), minEv, maxEv);
    x = (x - minEv) / (maxEv - minEv);

    const vec3 x2 = x * x;
    const vec3 x4 = x2 * x2;
    x = 15.5 * x4 * x2 - 40.14 * x4 * x
      + 31.96 * x4 - 6.868 * x2 * x
      + 0.4298 * x2 + 0.1191 * x - 0.00232;

    x = agxOut * x;
    return clamp(x, 0.0, 1.0);
}

vec3 ApplyTonemap(vec3 x)
{
    if (u_TonemapMode == 0) return TonemapACES(x);
    if (u_TonemapMode == 1) return TonemapReinhard(x);
    if (u_TonemapMode == 2) return TonemapFilmic(x);
    if (u_TonemapMode == 3) return TonemapAgX(x);
    return clamp(x, 0.0, 1.0);
}


vec2 DistortUV(vec2 uv)
{
    if (u_LensEnabled == 0)
        return uv;

    const vec2  c  = uv - 0.5;
    const float r2 = dot(c, c);
    return clamp(uv + c * r2 * u_LensDistortion * 4.0, vec2(0.0), vec2(1.0));
}

float Hash(vec2 p)
{
    return fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453);
}

void main()
{
    const vec2 uv = DistortUV(v_UV);

    
    const vec2  dir = uv - 0.5;
    const float ca  = (u_CAEnabled != 0) ? u_CAStrength * 0.01 : 0.0;

    vec3 hdr;
    hdr.r = texture(u_HDRTexture, clamp(uv + dir * ca, vec2(0.0), vec2(1.0))).r;
    hdr.g = texture(u_HDRTexture, uv).g;
    hdr.b = texture(u_HDRTexture, clamp(uv - dir * ca, vec2(0.0), vec2(1.0))).b;

    
    if (u_BloomEnabled != 0)
        hdr += texture(u_BloomTexture, uv).rgb * u_BloomIntensity;

    
    hdr = max(hdr * u_Exposure, vec3(0.0));

    vec3 color = ApplyTonemap(hdr);
    color = pow(color, vec3(1.0 / 2.2));

    
    
    color *= vec3(1.0 + 0.20 * u_Temperature,
                  1.0 + 0.10 * u_Tint,
                  1.0 - 0.20 * u_Temperature);

    
    color = color * u_Gain + u_Lift * (1.0 - color);
    color = pow(max(color, vec3(0.0)), 1.0 / max(u_Gamma, vec3(0.01)));

    
    color = (color - 0.5) * u_Contrast + 0.5;
    const float luma = dot(color, vec3(0.2126, 0.7152, 0.0722));
    color = mix(vec3(luma), color, u_Saturation);
    color = max(color, vec3(0.0));

    
    if (u_VignetteEnabled != 0)
    {
        const float d     = length(v_UV - 0.5) * 1.41421356;   
        const float inner = clamp(1.0 - u_VignetteSmoothness, 0.0, 1.0);
        const float vig   = smoothstep(inner, 1.0, d);
        color *= 1.0 - u_VignetteIntensity * vig;
    }

    
    if (u_GrainEnabled != 0)
    {
        const float n = Hash(v_UV * u_Resolution + vec2(u_Time * 137.0)) - 0.5;
        color += n * u_GrainIntensity;
    }

    FragColor = vec4(clamp(color, 0.0, 1.0), 1.0);
}
```
